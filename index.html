-------------------- FILE: lib/ratingEngine.ts -------------------- import { Team, Player, TeamAnalysis } from '@/types' // Position -> key attribute weights (0-1) const POSITION_WEIGHTS: Record> = { GK: { diving: 0.3, handling: 0.25, reflexes: 0.25, positioning: 0.2 }, LB: { pace: 0.3, defending: 0.3, stamina: 0.2, physical: 0.2 }, RB: { pace: 0.3, defending: 0.3, stamina: 0.2, physical: 0.2 }, dribbling: 0.2 }, CF: { shooting: 0.3, passing: 0.25, dribbling: 0.25, pace: 0.2 }, } // Map position to category for squad balance const POSITION_CATEGORY: Record = { GK: 'gk', LB: 'defense', RB: 'defense', CB: 'defense', CDM: 'midfield', CM: 'midfield', CAM: 'midfield', LM: 'midfield', RM: 'midfield', LW: 'attack', RW: 'attack', ST: 'attack', CF: 'attack' } // Helper: compute player quality (0-100) based on OVR, rank, training, and key attributes function computePlayerQuality(player: Player): number { // Base from OVR (scaled 0-100, FC Mobile OVR ~60-130) const ovrScore = Math.min(100, Math.max(0, (player.ovr - 60) / 70 * 100)) // Rank bonus: each rank level adds ~2 OVR equivalent const rankBonus = player.rankLevel * 2 // Training bonus: each training level adds ~0.5 OVR equivalent const trainBonus = player.trainingLevel * 0.5 // Attribute alignment: compute weighted average of relevant attributes const weights = POSITION_WEIGHTS[player.position] || {} let attrScore = 0 let totalWeight = 0 for (const [key, weight] of Object.entries(weights)) { const val = player.attributes[key as keyof typeof player.attributes] if (val !== undefined) { // attributes are 0-99 scale, normalize to 0-100 attrScore += (val / 99 * 100) * weight totalWeight += weight } } if (totalWeight === 0) attrScore = ovrScore // fallback else attrScore = attrScore / totalWeight // Combine: 50% OVR/rank/training, 50% attributes const baseQuality = (ovrScore + rankBonus + trainBonus) / (100 + 10 + 15) * 100 return baseQuality * 0.5 + attrScore * 0.5 } // Position penalty: if player plays out of natural position, reduce score function getPositionPenalty(player: Player): number { const natural = player.position // For simplicity, only ST at CB or GK outfield gets huge penalty if (natural === 'ST' && player.attributes.defending && player.attributes.defending < 30) return 30 if (natural === 'GK' && player.position !== 'GK') return 40 // GK playing outfield if (natural === 'CB' && player.attributes.pace && player.attributes.pace < 40) return 15 return 0 } // Main rating function export function analyzeTeam(team: Team): TeamAnalysis { const xi = team.startingXI const subs = team.substitutes || [] const reserves = team.reserves || [] // 1. Team OVR (25%) – but we also compute effective OVR from XI const avgXiOvr = xi.reduce((s, p) => s + p.ovr, 0) / xi.length const teamOvrScore = Math.min(100, (team.teamOvr - 60) / 70 * 100) // 2. Starting XI Player Quality (30%) let xiQualitySum = 0 const posScores: Record = {} for (const p of xi) { let q = computePlayerQuality(p) const penalty = getPositionPenalty(p) q = Math.max(0, q - penalty) posScores[p.id || p.name] = q xiQualitySum += q } const xiQualityAvg = xiQualitySum / xi.length // 3. Position Strength (15%) – penalize weak positions and out-of-position let positionStrengthSum = 0 for (const p of xi) { let score = computePlayerQuality(p) const penalty = getPositionPenalty(p) score = Math.max(0, score - penalty) // Also check if position is natural vs formation – for now use penalty positionStrengthSum += score } const positionStrength = positionStrengthSum / xi.length // 4. Key Attributes (10%) – already embedded in computePlayerQuality, but we re-weight // We'll recompute attribute-only score let attrSum = 0 for (const p of xi) { const weights = POSITION_WEIGHTS[p.position] || {} let aScore = 0, total = 0 for (const [key, w] of Object.entries(weights)) { const val = p.attributes[key as keyof typeof p.attributes] if (val !== undefined) { aScore += (val/99*100)*w; total += w } } attrSum += (total>0 ? aScore/total : 50) } const keyAttrScore = attrSum / xi.length // 5. Squad Balance (10%) – compute std dev of quality across categories const categories: Record = { attack: [], midfield: [], defense: [], gk: [] } for (const p of xi) { const cat = POSITION_CATEGORY[p.position] || 'midfield' const q = computePlayerQuality(p) - getPositionPenalty(p) categories[cat].push(q) } // Compute average per category const catAvgs: Record = {} for (const [cat, arr] of Object.entries(categories)) { catAvgs[cat] = arr.length ? arr.reduce((a,b) => a+b,0)/arr.length : 50 } // Balance = low variance between categories (normalized) const vals = Object.values(catAvgs) const mean = vals.reduce((a,b) => a+b,0)/vals.length const variance = vals.reduce((s,v) => s + (v-mean)**2, 0)/vals.length const balanceScore = Math.max(0, 100 - variance * 0.8) // high variance reduces balance // 6. Squad Depth (5%) – quality of subs and reserves let depthSum = 0 const allBench = [...subs, ...reserves] if (allBench.length > 0) { for (const p of allBench) { depthSum += computePlayerQuality(p) } const depthAvg = depthSum / allBench.length // Depth score is relative to XI average, cap at 100 depthScore = Math.min(100, (depthAvg / (xiQualityAvg || 1)) * 100) } else { depthScore = 50 // no subs = half } let depthScore = depthScore // 7. Team Synergy/Consistency (5%) – low std dev of XI quality const xiQ = xi.map(p => computePlayerQuality(p) - getPositionPenalty(p)) const xiMean = xiQ.reduce((a,b) => a+b,0)/xiQ.length const xiVar = xiQ.reduce((s,v) => s + (v-xiMean)**2, 0)/xiQ.length const consistencyScore = Math.max(0, 100 - xiVar * 0.5) // Weighted total let overall = teamOvrScore * 0.25 + xiQualityAvg * 0.30 + positionStrength * 0.15 + keyAttrScore * 0.10 + balanceScore * 0.10 + depthScore * 0.05 + consistencyScore * 0.05 overall = Math.min(100, Math.max(0, overall)) // Tier let tier = '' if (overall >= 95) tier = 'Elite' else if (overall >= 90) tier = 'World Class' else if (overall >= 85) tier = 'Excellent' else if (overall >= 80) tier = 'Strong' else if (overall >= 75) tier = 'Good' else if (overall >= 70) tier = 'Average' else if (overall >= 60) tier = 'Below Average' else tier = 'Weak' // Compute category scores for output const attackScore = Math.min(100, (catAvgs.attack || 50) * 1.0) const midfieldScore = Math.min(100, (catAvgs.midfield || 50) * 1.0) const defenseScore = Math.min(100, (catAvgs.defense || 50) * 1.0) const gkScore = Math.min(100, (catAvgs.gk || 50) * 1.0) // Weakness detection (top 3) const weaknesses: string[] = [] // Find lowest quality player in XI let minQ = Infinity, minP: Player | null = null for (const p of xi) { const q = computePlayerQuality(p) - getPositionPenalty(p) if (q < minQ) { minQ = q; minP = p } } if (minP) { const avg = xiQualityAvg const diff = avg - minQ if (diff > 10) weaknesses.push(`${minP.position} ${minP.name} is ${diff.toFixed(1)} points below squad average`) } // Check GK weakness const gkPlayers = xi.filter(p => p.position === 'GK') if (gkPlayers.length > 0) { const gkQ = computePlayerQuality(gkPlayers[0]) - getPositionPenalty(gkPlayers[0]) if (gkQ < 70) weaknesses.push(`Goalkeeper is weak (${gkQ.toFixed(1)})`) } // Check attack vs defense imbalance const attackAvg = catAvgs.attack || 0 const defAvg = catAvgs.defense || 0 if (attackAvg - defAvg > 15) weaknesses.push(`Attack (${attackAvg.toFixed(0)}) far outclasses defense (${defAvg.toFixed(0)})`) else if (defAvg - attackAvg > 15) weaknesses.push(`Defense (${defAvg.toFixed(0)}) far outclasses attack (${attackAvg.toFixed(0)})`) // Fill to 3 while (weaknesses.length < 3) { weaknesses.push('Squad depth could be improved') } weaknesses.splice(3) // Upgrade priority let upgradePriority = 'GK' let upgradeTarget = 'High OVR GK with good reflexes and handling' if (minP) { upgradePriority = minP.position const pos = minP.position const attrs = POSITION_WEIGHTS[pos] || {} const attrNames = Object.keys(attrs).slice(0,2).join(' and ') upgradeTarget = `${Math.ceil(minP.ovr + 5)}+ OVR ${pos} with high ${attrNames}` } // Conclusion let conclusion = `Your squad is ${tier.toLowerCase()}. ` if (attackScore > 80) conclusion += `Attack is strong (${attackScore.toFixed(0)}). ` else conclusion += `Attack needs improvement (${attackScore.toFixed(0)}). ` if (midfieldScore > 80) conclusion += `Midfield is solid (${midfieldScore.toFixed(0)}). ` else conclusion += `Midfield is a weak point (${midfieldScore.toFixed(0)}). ` if (defenseScore > 80) conclusion += `Defense is reliable (${defenseScore.toFixed(0)}). ` else conclusion += `Defense is below par (${defenseScore.toFixed(0)}). ` if (weaknesses.length) conclusion += `Priority: ${weaknesses[0]}.` return { overallScore: overall, tier, percentile: 0, // computed separately from DB attackScore, midfieldScore, defenseScore, gkScore, depthScore, balanceScore,
